```python
class TaskStatus(Enum):
    PENDING = 'pending'
    RUNNING = 'running'
    SUCCESS = 'success'
    FAILED = 'failed'
    PARTIAL = 'partial'

class KGTask(models.Model):
    """知识图谱任务模型"""
    TASK_TYPE_CHOICES = [
        ('build', '构建'),
        ('import', '导入'),
        ('export', '导出'),
        ('analyze', '分析'),
    ]
    
    STATUS_CHOICES = [
        ('pending', '待处理'),
        ('running', '运行中'),
        ('success', '成功'),
        ('failed', '失败'),
        ('partial', '部分成功'),
    ]
    
    task_id = models.CharField(max_length=36, unique=True, primary_key=True)
    task_type = models.CharField(max_length=20, choices=TASK_TYPE_CHOICES)
    status = models.CharField(max_length=20, choices=STATUS_CHOICES, default='pending')
    progress = models.IntegerField(default=0)  # 0-100
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    
    # 统计信息
    stats = models.JSONField(default=dict)  # {nodes_added, edges_added, skipped, conflicts}
    errors = models.JSONField(default=list)  # [{line, msg}]
    
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    completed_at = models.DateTimeField(null=True, blank=True)
    
    class Meta:
        ordering = ['-created_at']
        indexes = [
            models.Index(fields=['user', 'created_at']),
            models.Index(fields=['status']),
        ]
    
    def __str__(self):
        return f"{self.task_type}:{self.task_id}"
```

```python
    scope = request.data.get('scope')
    parameters = request.data.get('parameters', {})
    
    if not scope:
        return Response(
```

```python
from rest_framework.response import Response
from rest_framework import status
from datetime import datetime

class KGResponse:
    """知识图谱统一响应格式"""
    
    @staticmethod
    def success(data, pagination=None, meta=None, status_code=200):
        """成功响应"""
        response_data = {
            'data': data,
            'error': None,
        }
        
        if pagination:
            response_data['pagination'] = pagination
        
        if meta:
            response_data['meta'] = meta
        else:
            response_data['meta'] = {'query_time_ms': 0}
        
        return Response(response_data, status=status_code)
    
    @staticmethod
    def error(code, message, details=None, status_code=400):
        """错误响应"""
        response_data = {
            'data': None,
            'error': {
                'code': code,
                'message': message,
            }
        }
        
        if details:
            response_data['error']['details'] = details
        
        return Response(response_data, status=status_code)

class PaginationHelper:
    """分页辅助类"""
    
    SORT_WHITELIST = ['created_at', 'updated_at', 'weight', 'confidence']
    MAX_PAGE_SIZE = 1000
    DEFAULT_PAGE_SIZE = 20
    
    @staticmethod
    def validate_and_get_params(request):
        """验证并获取分页参数"""
        try:
            page = int(request.query_params.get('page', 1))
            page_size = int(request.query_params.get('page_size', 20))
            sort_by = request.query_params.get('sort_by', 'created_at')
            sort_order = request.query_params.get('sort_order', 'desc')
            
            # 验证
            if page < 1:
                page = 1
            if page_size < 1 or page_size > PaginationHelper.MAX_PAGE_SIZE:
                page_size = PaginationHelper.DEFAULT_PAGE_SIZE
            if sort_by not in PaginationHelper.SORT_WHITELIST:
                sort_by = 'created_at'
            if sort_order not in ['asc', 'desc']:
                sort_order = 'desc'
            
            return {
                'page': page,
                'page_size': page_size,
                'sort_by': sort_by,
                'sort_order': sort_order,
            }
        except (ValueError, TypeError):
            return None
    
    @staticmethod
    def get_pagination_meta(queryset, page, page_size):
        """获取分页元数据"""
        total = queryset.count()
        has_next = (page * page_size) < total
        
        return {
            'total': total,
            'page': page,
            'page_size': page_size,
            'has_next': has_next,
            'total_pages': (total + page_size - 1) // page_size,
        }
```

```python
    pagination_params = PaginationHelper.validate_and_get_params(request)
    if not pagination_params:

            'Invalid pagination parameters'
        )

    queryset = Node.objects.all()

    node_type = request.query_params.get('type')
    if node_type:

    # 应用排序
    sort_by = pagination_params['sort_by']
    sort_order = pagination_params['sort_order']
    order_field = f"{'-' if sort_order == 'desc' else ''}{sort_by}"
    queryset = queryset.order_by(order_field)
    
    # 分页
    page = pagination_params['page']
    page_size = pagination_params['page_size']
    offset = (page - 1) * page_size
    
    total = queryset.count()
    items = queryset[offset:offset + page_size]
    
    # 序列化
    serializer = NodeSerializer(items, many=True)
    
    # 构建响应
    query_time = int((time.time() - start_time) * 1000)
    pagination_meta = PaginationHelper.get_pagination_meta(queryset, page, page_size)
    
    return KGResponse.success(
        serializer.data,
        pagination=pagination_meta,
        meta={'query_time_ms': query_time}
```

```python
class RecommendationService:
    
    @staticmethod
    def suggest_edges(node_id, top_k=10):
        """推荐候选边"""
        suggestions = []
        
        # 基于共现的推荐
        cooccurrence_edges = RecommendationService._get_cooccurrence_edges(node_id)
        for edge in cooccurrence_edges[:top_k]:
            suggestions.append({
                'source': node_id,
                'target': edge['target_id'],
                'type': edge['type'],
                'confidence': edge['confidence'],
                'evidence': [
                    {
                        'type': 'cooccurrence',
                        'score': edge['confidence'],
                        'detail': f"共现于 {edge['count']} 条笔记"
                    }
                ]
            })
        
        return suggestions
    
    @staticmethod
    def accept_suggestions(edge_ids):
        """采纳建议"""
        accepted = 0
        errors = []
        
        for edge_id in edge_ids:
            try:
                # 创建边
                edge = Edge.objects.create(
                    source_id=edge_id['source'],
                    target_id=edge_id['target'],
                    edge_type=edge_id['type'],
                    confidence=edge_id.get('confidence', 1.0),
                    is_suggested=False,  # 标记为已采纳
                )
                
                # 记录建议
                SuggestionRecord.objects.create(
                    source_id=edge_id['source'],
                    target_id=edge_id['target'],
                    action='accepted',
                )
                
                accepted += 1
            except Exception as e:
                errors.append({
                    'edge_id': edge_id,
                    'error': str(e)
                })
        
        return {'accepted': accepted, 'errors': errors}
```

```python
class GraphService:
    
    AGGREGATION_THRESHOLD = 1000
    
    @staticmethod
    def get_nodes(filters=None, aggregate=False):
        """获取节点，支持聚合"""
        queryset = Node.objects.all()
        
        # 应用过滤
        if filters:
            queryset = GraphService._apply_filters(queryset, filters)
        
        node_count = queryset.count()
        
        # 检查是否需要聚合
        if aggregate and node_count > GraphService.AGGREGATION_THRESHOLD:
            return GraphService._aggregate_nodes(queryset)
        
        return {
            'nodes': NodeSerializer(queryset, many=True).data,
            'aggregation_info': {
                'original_count': node_count,
                'aggregated_count': node_count,
                'strategy': 'none'
            }
        }
    
    @staticmethod
    def _aggregate_nodes(queryset):
        """聚合节点"""
        # 按类型聚合
        aggregated = {}
        for node in queryset:
            key = node.node_type
            if key not in aggregated:
                aggregated[key] = {
                    'type': key,
                    'count': 0,
                    'nodes': []
                }
            aggregated[key]['count'] += 1
            aggregated[key]['nodes'].append(node.id)
        
        # 转换为节点列表
        aggregated_nodes = []
        for key, group in aggregated.items():
            aggregated_nodes.append({
                'id': f"agg_{key}",
                'label': f"{key} ({group['count']})",
                'node_type': key,
                'is_aggregated': True,
                'child_nodes': group['nodes'],
                'child_count': group['count'],
            })
        
        return {
            'nodes': aggregated_nodes,
            'aggregation_info': {
                'original_count': queryset.count(),
                'aggregated_count': len(aggregated_nodes),
                'strategy': 'by_type'
            }
        }
```
